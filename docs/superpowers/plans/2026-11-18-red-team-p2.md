```python
"""Confere que o P2 ficou como estava: hash do corpus, commit e git status."""

from __future__ import annotations

import hashlib
import subprocess
from pathlib import Path


class P2Mudou(Exception):
    """O P2 mudou durante a rodada: ela é inválida."""


def corpus_sha256(base: Path) -> str:
    return hashlib.sha256((Path(base) / "corpus" / "dispositivos.json").read_bytes()).hexdigest()


def _git(p2_dir: Path, *args: str) -> str:
    return subprocess.run(["git", "-C", str(p2_dir), *args],
                          capture_output=True, text=True, check=True).stdout


def snapshot(p2_dir: Path) -> dict:
    return {
        "corpus_sha256": corpus_sha256(p2_dir),
        "git_head": _git(p2_dir, "rev-parse", "HEAD").strip(),
        "git_status": _git(p2_dir, "status", "--porcelain"),
    }


def conferir(antes: dict, depois: dict) -> None:
    diferentes = [k for k in antes if antes[k] != depois.get(k)]
    if diferentes:
        raise P2Mudou(f"o P2 mudou durante a rodada: {', '.join(diferentes)}")
```

```python
"""Cópia de trabalho do P2: quem mexe aqui é o laboratório, nunca o original."""

from __future__ import annotations

import json
import shutil
from pathlib import Path

MCP_CONFIG_NAME = "redteam-mcp.json"
_IGNORAR = shutil.ignore_patterns(".git", ".venv", "traces", "__pycache__", ".pytest_cache")


def prepare_workdir(p2_dir: Path, dest: Path) -> Path:
    """Copia o checkout do P2 e grava a config MCP só com o server taxlaw.

    O copytree levanta FileExistsError se o destino já existe: uma cópia nunca é reaproveitada.
    """
    cfg = json.loads((Path(p2_dir) / ".mcp.json").read_text(encoding="utf-8"))["mcpServers"]["taxlaw"]
    shutil.copytree(p2_dir, dest, ignore=_IGNORAR)
    (Path(dest) / MCP_CONFIG_NAME).write_text(
        json.dumps({"mcpServers": {"taxlaw": cfg}}, indent=2), encoding="utf-8")
    return Path(dest)
```

```python
"""Esquema comum do catálogo de ataques: um arquivo JSON por ataque.

Valida só os campos comuns. Cada laboratório valida o próprio bloco de entrada e de critério.
"""

from __future__ import annotations

import json
import re
from dataclasses import dataclass
from pathlib import Path

CATALOGO_DIR = Path(__file__).parent / "ataques"
SUPERFICIES = {"injecao": "I", "http": "H"}
CAMPOS = ("id", "superficie", "nome", "nucleo", "rotulos", "previsao")
ID_RE = re.compile(r"[IH][1-9][0-9]?")


@dataclass(frozen=True)
class Attack:
    id: str
    superficie: str
    nome: str
    nucleo: bool
    rotulos: dict
    previsao: dict


def _erro(caminho: Path, msg: str):
    raise ValueError(f"{caminho.name}: {msg}")


def load_attack(caminho: Path) -> Attack:
    caminho = Path(caminho)
    d = json.loads(caminho.read_text(encoding="utf-8"))
    for campo in CAMPOS:
        if campo not in d:
            _erro(caminho, f"sem o campo {campo!r}")
    if not isinstance(d["id"], str) or not ID_RE.fullmatch(d["id"]):
        _erro(caminho, f"id inválido: {d['id']!r}")
    if d["superficie"] not in SUPERFICIES:
        _erro(caminho, f"superficie inválida: {d['superficie']!r}")
    if d["id"][0] != SUPERFICIES[d["superficie"]]:
        _erro(caminho, "o prefixo do id não combina com a superfície")
    if caminho.stem != d["id"]:
        _erro(caminho, "o nome do arquivo deve ser o id do ataque")
    if not isinstance(d["nucleo"], bool):
        _erro(caminho, "nucleo deve ser true ou false")
    if not isinstance(d["rotulos"], dict) or not isinstance(d["previsao"], dict):
        _erro(caminho, "rotulos e previsao devem ser objetos")
    return Attack(**{c: d[c] for c in CAMPOS})


def load_catalog(diretorio: Path = CATALOGO_DIR) -> list[Attack]:
    ataques = [load_attack(p) for p in sorted(Path(diretorio).glob("*.json"))]
    return sorted(ataques, key=lambda a: (a.superficie != "injecao", int(a.id[1:])))
```

`redteam/estatistica.py`:

```python
"""Taxas com intervalo de Wilson de 95%: seguro mesmo com poucas execuções e com 0 ou 100%."""

from __future__ import annotations

import math


def wilson(k: int, n: int, z: float = 1.959964) -> tuple[float, float]:
    if n < 0 or k < 0 or k > n:
        raise ValueError(f"k e n inválidos: k={k}, n={n}")
    if n == 0:
        return (0.0, 1.0)
    p = k / n
    z2 = z * z
    centro = (p + z2 / (2 * n)) / (1 + z2 / n)
    meio = z * math.sqrt(p * (1 - p) / n + z2 / (4 * n * n)) / (1 + z2 / n)
    return (max(0.0, centro - meio), min(1.0, centro + meio))


def taxa(k: int, n: int) -> str:
    if n == 0:
        return "0/0 (sem execuções)"
    a, b = wilson(k, n)
    return f"{k}/{n} = {round(100 * k / n)}% (IC 95%: {round(100 * a)}% a {round(100 * b)}%)"
```

## Tarefa 5: Varredura de números de contribuinte no repositório

**Para estudar:** a regra "nenhum CPF no repositório" só vale se algo a faz cumprir. Esta varredura reprova o commit que a viola. Um achado exige formato **e** dígito verificador válido, como na guarda do P2. Sem o DV, qualquer sequência de 11 dígitos (um número de execução do GitHub, por exemplo) seria acusada. Falsos positivos entram numa lista de exceções por *fingerprint*, de modo que a lista também não contém o número. Esta versão cobre CPF e CNPJ, formatados e puros; o CNPJ alfanumérico em minúsculas não é coberto, e isso fica registrado como limite.

**Arquivos:** criar `redteam/varredura.py`, `redteam/varredura_allow.txt`; teste `redteam/tests/test_varredura.py`.

**Produz:** `valido(tipo, numero) -> bool`, `fingerprint(numero) -> str`, `scan_text(texto) -> list[tuple[int, str, str]]`, `scan_file(caminho, allow) -> list[Hit]`, `main(argv) -> int`, e o comando `python -m redteam.varredura`.

```python
"""Varredura: nenhum CPF ou CNPJ com dígito verificador válido nos arquivos rastreados pelo git.

Um achado mostra caminho, linha, tipo e fingerprint, nunca o número. Falsos positivos entram
em `redteam/varredura_allow.txt` pelo fingerprint.
"""

from __future__ import annotations

import argparse
import hashlib
import re
import subprocess
import sys
from dataclasses import dataclass
from pathlib import Path

_ANTES = r"(?<![0-9A-Za-z])"
_DEPOIS = r"(?![0-9A-Za-z])"
_REGEX = [
    ("cpf", re.compile(_ANTES + r"[0-9]{3}\.[0-9]{3}\.[0-9]{3}-[0-9]{2}" + _DEPOIS)),
    ("cpf", re.compile(_ANTES + r"[0-9]{11}" + _DEPOIS)),
    ("cnpj", re.compile(_ANTES + r"[0-9A-Z]{2}\.[0-9A-Z]{3}\.[0-9A-Z]{3}/[0-9A-Z]{4}-[0-9]{2}" + _DEPOIS)),
    ("cnpj", re.compile(_ANTES + r"[0-9A-Z]{14}" + _DEPOIS)),
]
_SEP = re.compile(r"[./-]")
CPF_P1 = tuple(range(10, 1, -1))
CPF_P2 = tuple(range(11, 1, -1))
CNPJ_P1 = (5, 4, 3, 2, 9, 8, 7, 6, 5, 4, 3, 2)
CNPJ_P2 = (6, 5, 4, 3, 2, 9, 8, 7, 6, 5, 4, 3, 2)
MAX_BYTES = 5_000_000
ALLOW_PADRAO = Path(__file__).parent / "varredura_allow.txt"


@dataclass(frozen=True)
class Hit:
    caminho: str
    linha: int
    tipo: str
    fingerprint: str


def _dv(valores: list[int], pesos: tuple[int, ...]) -> int:
    r = sum(v * p for v, p in zip(valores, pesos, strict=True)) % 11
    return 0 if r < 2 else 11 - r


def valido(tipo: str, numero: str) -> bool:
    if tipo == "cpf":
        if not re.fullmatch(r"[0-9]{11}", numero) or len(set(numero)) == 1:
            return False
        base = [int(c) for c in numero[:9]]
        d1 = _dv(base, CPF_P1)
        return numero[9:] == f"{d1}{_dv(base + [d1], CPF_P2)}"
    if not re.fullmatch(r"[0-9A-Z]{14}", numero) or len(set(numero)) == 1 or not numero[12:].isdigit():
        return False
    v = [ord(c) - 48 for c in numero[:12]]
    d1 = _dv(v, CNPJ_P1)
    return numero[12:] == f"{d1}{_dv(v + [d1], CNPJ_P2)}"


def fingerprint(numero: str) -> str:
    return hashlib.sha256(numero.encode()).hexdigest()[:12]


def scan_text(texto: str) -> list[tuple[int, str, str]]:
    achados = []
    for n_linha, linha in enumerate(texto.splitlines(), 1):
        for tipo, regex in _REGEX:
            for m in regex.finditer(linha):
                compacto = _SEP.sub("", m.group()).upper()
                if valido(tipo, compacto):
                    achados.append((n_linha, tipo, fingerprint(compacto)))
    return achados


def carregar_allow(caminho: Path = ALLOW_PADRAO) -> set[str]:
    if not caminho.exists():
        return set()
    entradas = set()
    for linha in caminho.read_text(encoding="utf-8").splitlines():
        linha = linha.split("#", 1)[0].strip()
        if linha:
            entradas.add(linha)
    return entradas


def scan_file(caminho: Path, allow: set[str]) -> list[Hit]:
    try:
        if caminho.stat().st_size > MAX_BYTES:
            return []
        texto = caminho.read_bytes().decode("utf-8")
    except (OSError, UnicodeDecodeError):
        return []
    return [Hit(str(caminho), n, tipo, fp) for n, tipo, fp in scan_text(texto) if fp not in allow]


def arquivos_rastreados(raiz: Path) -> list[Path]:
    saida = subprocess.run(["git", "-C", str(raiz), "ls-files", "-z"],
                           capture_output=True, check=True).stdout.decode()
    return [Path(raiz) / p for p in saida.split("\0") if p]


def main(argv: list[str] | None = None) -> int:
    ap = argparse.ArgumentParser(description="Procura CPF e CNPJ válidos nos arquivos do repositório.")
    ap.add_argument("--raiz", type=Path, default=Path("."))
    ap.add_argument("--allow", type=Path, default=ALLOW_PADRAO)
    ap.add_argument("--arquivos", nargs="*", type=Path, default=None)
    args = ap.parse_args(argv)
    allow = carregar_allow(args.allow)
    arquivos = args.arquivos if args.arquivos is not None else arquivos_rastreados(args.raiz)
    achados = [h for a in arquivos for h in scan_file(a, allow)]
    for h in achados:
        print(f"{h.caminho}:{h.linha}: {h.tipo} com dígito verificador válido (fingerprint {h.fingerprint})")
    if achados:
        print(f"{len(achados)} achado(s): remova o número ou, se for falso positivo, "
              f"acrescente o fingerprint em {args.allow.name}.", file=sys.stderr)
        return 1
    print("nenhum CPF ou CNPJ válido encontrado")
    return 0


if __name__ == "__main__":
    raise SystemExit(main())
```
